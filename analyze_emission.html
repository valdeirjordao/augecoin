// ===============================

use crate::constants::{
    CT_FIRST_REWARD_AUGESAT, CT_MIN_REWARD_AUGESAT, CT_NEW_LINE_REWARD_DECREASE,
    CT_DEVELOPER_ACCOUNT_START, CT_DEVELOPER_ACCOUNT_COUNT, CT_DEVELOPER_REWARD_PERCENT,
};
use crate::constants::ONE_AUGE;

pub const TOTAL_SUPPLY_AUGE: u64 = 750_000_000;
pub const TOTAL_SUPPLY_AUGESAT: u64 = TOTAL_SUPPLY_AUGE.saturating_mul(ONE_AUGE);
                             // = 75.000.000.000.000.000

// ═══════════════════════════════════════════════════════
//  block_reward — recompensa com halving exponencial
// ═══════════════════════════════════════════════════════
pub fn block_reward(block_number: u64) -> u64 {
    if block_number == 0 {
        return CT_FIRST_REWARD_AUGESAT;             // bloco gênesis
    }

    let halvings = block_number / CT_NEW_LINE_REWARD_DECREASE;  // a cada 210.240 blocos
    let mut reward = CT_FIRST_REWARD_AUGESAT;
    for _ in 0..halvings {
        reward /= 2;
        if reward < CT_MIN_REWARD_AUGESAT {
            return CT_MIN_REWARD_AUGESAT;              // piso: 0,1 AUGE
        }
    }
    if reward < CT_MIN_REWARD_AUGESAT {
        CT_MIN_REWARD_AUGESAT
    } else {
        reward
    }
}

// ═══════════════════════════════════════════════════════
//  developer_reward — 20% da recompensa por bloco
// ═══════════════════════════════════════════════════════
pub fn developer_reward(reward: u64) -> u64 {
    reward.saturating_mul(CT_DEVELOPER_REWARD_PERCENT) / 100
}

// ═══════════════════════════════════════════════════════
//  developer_leader_reward — 80% para o líder (miner)
// ═══════════════════════════════════════════════════════
pub fn developer_leader_reward(reward: u64) -> u64 {
    reward.saturating_sub(developer_reward(reward))
}

// ═══════════════════════════════════════════════════════
//  developer_account_for_block — rotação entre devs
// ═══════════════════════════════════════════════════════
pub fn developer_account_for_block(block_number: u64) -> u64 {
    if CT_DEVELOPER_ACCOUNT_COUNT == 0 { return 0; }
    CT_DEVELOPER_ACCOUNT_START + (block_number % CT_DEVELOPER_ACCOUNT_COUNT)
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn genesis_block_reward() {
        assert_eq!(block_reward(0), CT_FIRST_REWARD_AUGESAT);
    }

    #[test]
    fn first_block_reward() {
        assert_eq!(block_reward(1), CT_FIRST_REWARD_AUGESAT);
    }

    #[test]
    fn before_first_halving() {
        for h in [1, 100, 1000, 100000, CT_NEW_LINE_REWARD_DECREASE - 1] {
            assert_eq!(block_reward(h), CT_FIRST_REWARD_AUGESAT);
        }
    }

    #[test]
    fn after_first_halving() {
        let halving_block = CT_NEW_LINE_REWARD_DECREASE;
        let expected = CT_FIRST_REWARD_AUGESAT / 2;
        assert_eq!(block_reward(halving_block), expected);
        assert_eq!(block_reward(halving_block + 1), expected);
    }

    #[test]
    fn after_second_halving() {
        let h = CT_NEW_LINE_REWARD_DECREASE * 2;
        assert_eq!(block_reward(h), CT_FIRST_REWARD_AUGESAT / 4);
    }

    #[test]
    fn bottoms_out_at_min_reward() {
        let far = CT_NEW_LINE_REWARD_DECREASE * 100;
        assert_eq!(block_reward(far), CT_MIN_REWARD_AUGESAT);
    }

    #[test]
    fn developer_reward_is_20_percent() {
        let reward = 100_000;
        let dev = developer_reward(reward);
        let leader = developer_leader_reward(reward);
        assert_eq!(dev, 20_000);
        assert_eq!(leader, 80_000);
        assert_eq!(dev + leader, reward);
    }

    #[test]
    fn developer_account_rotation() {
        let dev_count = CT_DEVELOPER_ACCOUNT_COUNT;
        for i in 0..dev_count * 3 {
            let expected = CT_DEVELOPER_ACCOUNT_START + (i % dev_count);
            assert_eq!(developer_account_for_block(i), expected);
        }
    }
}

pub fn execute_block(
    block: &OperationBlock,
    storage: &Storage,
) -> Result<[u8; 64], ExecutionError> {
    let block_number = block.header.block_number;
    let leader_id = block.header.leader_id;

    // Snapshot para atomicidade
    let mut modified: HashMap<u64, Account> = HashMap::new();
    let mut original: HashMap<u64, Account> = HashMap::new();

    // ... load_account closure ...

    // ── FASE 1: Verificação paralela de assinaturas ──
    let verification_results = block.operations.par_iter()
        .enumerate()
        .map(|(idx, op)| verify_operation_signatures(op, storage, block_number)
            .map_err(|e| (idx, e)))
        .collect::<Vec<_>>();

    // ── FASE 2: Execução sequencial das operações ──
    let total_fee_ref = &mut 0u64;
    for op in &block.operations {
        let op_fee = execute_operation(op, block_number, &mut modified,
                                      &mut original, storage)?;
        *total_fee_ref = total_fee_ref.saturating_add(op_fee);
    }
    let total_fees = *total_fee_ref;

    // ── FASE 3: Distribuir fees + block reward ──
    let reward = block_reward(block_number);          // ★ CHAMADA CENTRAL
    let dev_reward = developer_reward(reward);       // 20%
    let leader_reward = developer_leader_reward(reward); // 80%

    // Developer recebe 20% do block reward + todas as fees
    // Leader recebe 80% do block reward
    let dev_account_num = developer_account_for_block(block_number);

    if dev_account_num == leader_id {
        // Mesma conta: leader = dev — recebe reward integral + fees
        leader_account.add_balance(reward.saturating_add(total_fees))?;
        // ... cria CT_ACCOUNTS_PER_BLOCK novas contas com leader_key ...
    } else {
        // Contas separadas
        dev_account.add_balance(dev_reward.saturating_add(total_fees))?;
        leader_account.add_balance(leader_reward)?;
        // ... cria CT_ACCOUNTS_PER_BLOCK novas contas com leader_key ...
    }

    // ── FASE 5: Compute novo SafeBox hash ──
    let mut safebox = build_safebox_from_modified(storage, &modified)?;
    safebox.update_hash();

    // ── FASE 6: Commit atômico ──
    commit_changes(&modified, storage)?;
    storage.put_safe_box(&safebox)?;

    Ok(safe_box_hash)
}
